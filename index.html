    03 / CÓDIGO EXPLICADO
    
      
    Por qué funciona.
    
      
    
        01 — CSS: del layout rígido al componente flexible
        
    .sessions { display: grid; grid-template-columns:
  repeat(auto-fit, minmax(min(100%, 18rem), 1fr)); }
.card { min-width: 0; }
.card__title { overflow-wrap: anywhere; }
.card-slot { container-type: inline-size; }
@container (min-width: 24rem) {
  .card { grid-template-columns: 4.5rem minmax(0, 1fr); }
}
    
      
    
      
    
        02 — JS: no interpretar contenido no confiable como HTML
        
    const heading = document.createElement('h3');
heading.textContent = session.title ?? 'Sin título';
const message = document.querySelector('#announce');
message.textContent = 'Contenido actualizado';
    
      
    
      
    
        03 — A11y: cambios de estado sin mover el foco
        
    <p id="announce" role="status" aria-atomic="true"></p>

@media (prefers-reduced-motion: reduce) {
  *, *::before, *::after { scroll-behavior: auto; }
}
